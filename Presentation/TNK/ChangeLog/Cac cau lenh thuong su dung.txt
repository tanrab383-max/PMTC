select *
from ChiTietBaoDuongTietKiem
where MaPhieuThu in
(select MaPhieuThu
from ChiTietBaoDuongTietKiem 
group by MaPhieuThu having count(*) = 1
)

commit transaction
declare cur cursor for
select MaPhieuThu
from ChiTietBaoDuongTietKiem 
group by MaPhieuThu having count(*) = 1

declare @maPhieuThu varchar(20), @sl int
declare @index int = 1
open cur
fetch next from cur into @maPhieuThu,@sl
while(@@FETCH_STATUS = 0)
begin
	print cast(@index as varchar(100))
	print @maPhieuThu
	declare @ngayThu datetime
	select @ngayThu = NgayThu from PhieuThu where MaPhieuThu = @maPhieuThu and IsDeleted = 0
	update ChiTietBaoDuongTietKiem 
	set HanSuDung = dateadd(month,7,@ngayThu) 
	where MaPhieuThu = @maPhieuThu and HanSuDung is null
	set @index = @index + 1
	fetch next from cur into @maPhieuThu,@sl
end
close cur
deallocate cur
